DNSTestToolkit v2.0 — Overview & Architecture
Pluggable PowerShell test engine for validating BlueCat MDDS, Edge Resolver, and Anycast DNS infrastructure across AWS, Azure, GCP, and Equinix Fabric. v2.0 adds five new modules covering zone transfers, threat protection, sinkhole validation, TSIG dynamic updates, and advanced ECS cross-cloud CNAME steering.
All scripts target PowerShell 7.4+ (cross-platform). Run from any Windows or Linux jump host with network path to DNS VIPs and resolver targets. Designed to run concurrently from multiple regions to simulate real distributed client load.

Engine Core — DNSTestEngine.ps1
Auto-discovering module loader, parallel query executor, percentile calculator, and threshold evaluator. Unchanged from v1.0 — all new modules use the same contract.
DNSTestEngine.ps1 
# ================================================================
# DNSTestEngine.ps1 — DNSTestToolkit v2.0
# Core engine: module auto-discovery, parallel queries,
# percentile math, threshold evaluation, result dispatch
# ================================================================

$script:ENGINE_VERSION = '2.0'
$script:MODULE_DIR     = Join-Path $PSScriptRoot 'modules'
$script:CONFIG_DIR     = Join-Path $PSScriptRoot 'config'
$script:REPORT_DIR     = Join-Path $PSScriptRoot 'reports'
$script:RunId          = [guid]::NewGuid().ToString('N').Substring(0,8).ToUpper()
$script:LoadedModules  = [ordered]@{}

function Write-EngineLog($Level, $Msg) {
    $color = switch($Level) {
        'INFO'  { 'Cyan'    } 'PASS' { 'Green'  }
        'WARN'  { 'Yellow' } 'FAIL' { 'Red'    }
        'DEBUG' { 'Gray'   } default { 'White'  }
    }
    Write-Host "[$(Get-Date -f 'HH:mm:ss')] [ENGINE] [$Level] $Msg" -ForegroundColor $color
}

function Get-EngineConfig([string]$Name) {
    $path = Join-Path $script:CONFIG_DIR "$Name.json"
    if (-not (Test-Path $path)) { throw "Config not found: $path" }
    Get-Content $path -Raw | ConvertFrom-Json
}

function Import-TestModules {
    param([string[]]$ModuleFilter = @('*'))
    Get-ChildItem $script:MODULE_DIR -Filter 'DNSTest-*.psm1' | ForEach-Object {
        try {
            Import-Module $_.FullName -Force -Global -ErrorAction Stop
            $info = Get-ModuleInfo
            if ($ModuleFilter -contains '*' -or $ModuleFilter -contains $info.Id -or
                ($info.Tags | Where-Object { $ModuleFilter -contains $_ })) {
                $script:LoadedModules[$info.Id] = $info
                Write-EngineLog DEBUG "Loaded: $($info.Id) — $($info.Name) v$($info.Version)"
            }
        } catch { Write-EngineLog WARN "Failed to load $($_.Name): $_" }
    }
    Write-EngineLog INFO "$($script:LoadedModules.Count) modules loaded"
}

function Invoke-DNSQuery {
    param(
        [string]  $Server,
        [string]  $Name,
        [string]  $Type       = 'A',
        [int]     $TimeoutMs  = 3000,
        [switch]  $UseTCP,
        [string]  $ECSSubnet  = ''
    )
    $sw = [System.Diagnostics.Stopwatch]::StartNew()
    try {
        $digArgs = @("@$Server", $Name, $Type, '+time=3', '+tries=1', '+short')
        if ($UseTCP)   { $digArgs += '+tcp' }
        if ($ECSSubnet) { $digArgs += "+subnet=$ECSSubnet" }
        $out    = & dig @digArgs 2>&1
        $status = if ($out -match 'SERVFAIL|REFUSED|NXDOMAIN|FORMERR') {
            [regex]::Match($out,'SERVFAIL|REFUSED|NXDOMAIN|FORMERR').Value
        } elseif ($out.Count -gt 0) { 'NOERROR' } else { 'TIMEOUT' }
        $sw.Stop()
        [pscustomobject]@{
            Server    = $Server; Name = $Name; Type = $Type
            Status    = $status; LatencyMs = [math]::Round($sw.Elapsed.TotalMilliseconds,2)
            Answer    = $out; ECSSubnet = $ECSSubnet
        }
    } catch {
        $sw.Stop()
        [pscustomobject]@{ Server=$Server; Name=$Name; Type=$Type; Status='ERROR'
            LatencyMs=$sw.Elapsed.TotalMilliseconds; Answer=@($_.ToString()) }
    }
}

function Invoke-ParallelDNS {
    param([array]$Queries, [int]$Throttle = 50)
    $Queries | ForEach-Object -Parallel {
        $q = $_
        $sw = [System.Diagnostics.Stopwatch]::StartNew()
        $args = @("@$($q.Server)", $q.Name, $q.Type, '+time=3', '+tries=1', '+short')
        if ($q.ECSSubnet) { $args += "+subnet=$($q.ECSSubnet)" }
        $out = & dig @args 2>&1
        $sw.Stop()
        [pscustomobject]@{
            Server=$q.Server; Name=$q.Name; Type=$q.Type
            LatencyMs=[math]::Round($sw.Elapsed.TotalMilliseconds,2)
            Status=if($out -match 'NXDOMAIN|SERVFAIL|REFUSED'){'BLOCKED'}elseif($out){'NOERROR'}else{'TIMEOUT'}
            Answer=$out
        }
    } -ThrottleLimit $Throttle
}

function Get-Percentile([double[]]$Values, [int]$Pct) {
    if (-not $Values) { return 0 }
    $sorted = $Values | Sort-Object
    $idx    = [math]::Ceiling(($Pct / 100) * $sorted.Count) - 1
    $sorted[[math]::Max(0, $idx)]
}

function Get-LatencyStats([double[]]$Values) {
    [pscustomobject]@{
        Count  = $Values.Count
        MinMs  = [math]::Round(($Values | Measure-Object -Minimum).Minimum, 2)
        MaxMs  = [math]::Round(($Values | Measure-Object -Maximum).Maximum, 2)
        AvgMs  = [math]::Round(($Values | Measure-Object -Average).Average,  2)
        P50Ms  = [math]::Round((Get-Percentile $Values 50), 2)
        P95Ms  = [math]::Round((Get-Percentile $Values 95), 2)
        P99Ms  = [math]::Round((Get-Percentile $Values 99), 2)
    }
}

function Test-Threshold($Value, $PassThreshold, $WarnThreshold, [switch]$LowerIsBetter = $true) {
    if ($LowerIsBetter) {
        if ($Value -le $PassThreshold) { return 'PASS' }
        if ($Value -le $WarnThreshold) { return 'WARN' }
        return 'FAIL'
    } else {
        if ($Value -ge $PassThreshold) { return 'PASS' }
        if ($Value -ge $WarnThreshold) { return 'WARN' }
        return 'FAIL'
    }
}

function Add-TestResult([hashtable]$Result) {
    $Result.Timestamp = Get-Date -Format 'o'
    $script:Results  += [pscustomobject]$Result
}

function Invoke-TestEngine {
    param(
        [string[]]  $Modules   = @('*'),
        [string]    $Scenario = 'full',
        [switch]    $NoReport,
        [switch]    $DryRun
    )
    Write-EngineLog INFO "═══ DNSTestToolkit v$script:ENGINE_VERSION · Run $script:RunId ═══"
    $config     = Get-EngineConfig 'targets'
    $thresholds = Get-EngineConfig 'thresholds'
    $zones      = Get-EngineConfig 'zones'
    $tsigKeys   = Get-EngineConfig 'tsig-keys'
    $threatFeeds= Get-EngineConfig 'threat-feeds'
    Import-TestModules -ModuleFilter $Modules
    $runDir = New-Item -ItemType Directory -Force `
        -Path (Join-Path $script:REPORT_DIR "$(Get-Date -f 'yyyyMMdd-HHmmss')-$script:RunId")
    New-Item -ItemType Directory -Path (Join-Path $runDir 'raw') -Force | Out-Null
    foreach ($id in $script:LoadedModules.Keys) {
        if ($DryRun) { Write-EngineLog DEBUG "[DRY-RUN] $id"; continue }
        $ctx = @{
            Config=$config; Thresholds=$thresholds; Zones=$zones
            TSIGKeys=$tsigKeys; ThreatFeeds=$threatFeeds
            RunId=$script:RunId; RunDir=$runDir.FullName; Scenario=$Scenario
            FnQuery=$function:Invoke-DNSQuery; FnParallel=$function:Invoke-ParallelDNS
            FnStats=$function:Get-LatencyStats; FnPercentile=$function:Get-Percentile
            FnThreshold=$function:Test-Threshold; FnAddResult=$function:Add-TestResult
        }
        try {
            $r = Invoke-DNSTest -Context $ctx
            $r | ConvertTo-Json -Depth 10 | Out-File (Join-Path $runDir "raw/$id.json") -Force
            Write-EngineLog PASS "$id — $($r.Summary.Status)"
        } catch { Write-EngineLog FAIL "$id threw: $_" }
    }
    if (-not $NoReport) { Invoke-DNSTestReport -RunDir $runDir.FullName }
    Write-EngineLog INFO "Run $script:RunId complete — $($runDir.FullName)/index.html"
}

New Modules — v2.0
Five new pluggable test modules added in v2.0: zone transfers, threat protection, DNS sinkholes, TSIG dynamic updates, and advanced ECS cross-cloud CNAME steering.
All new modules follow the same contract as existing modules. They are auto-discovered at engine startup and can be combined with original modules in any scenario playbook.
MOD-10 Zone Transfer — AXFR/IXFR
Validates AXFR and IXFR zone transfers from standalone Windows DNS servers for external and split-horizon zones. Tests ACL enforcement, SOA serial comparison, record count integrity, and secondary sync latency.
AXFR from Windows DNS servers requires the test host IP to be listed in the zone's Transfer ACL. Use dnscmd /ZoneInfo <zone> on the Windows DNS server to verify. IXFR tests require a known prior serial — supply via config or let the module perform an initial SOA query.
DNSTest-ZoneTransfer.psm1 — MOD-10 
# ============================================================
# DNSTest-ZoneTransfer.psm1  –  MOD-10 · Zone Transfer Tests
# AXFR/IXFR from standalone Windows DNS (external + split-DNS)
# ============================================================

function Get-ModuleInfo {
    [pscustomobject]@{
        Id='MOD-10'; Name='Zone Transfer (AXFR/IXFR)'; Version='2.0'
        Tags=@('zone-transfer','axfr','ixfr','windows-dns','split-dns','security')
    }
}

function Get-DefaultConfig {
    @{
        # Transfer types to exercise
        TransferTypes        = @('AXFR', 'IXFR')

        # Max acceptable record-count delta between primary and secondary (%)
        RecordDeltaThreshold = 0.01   # 1% — any more = data integrity issue

        # Max seconds a secondary is allowed to lag behind primary SOA serial
        SerialLagThresholdSec = 300   # 5 min

        # Expected REFUSED from unauthorized source IPs
        TestAclBlocking      = $true

        # IP that is NOT in the transfer ACL — engine queries from this via loopback trick
        UnauthorizedProbeIP  = '127.0.0.2'

        TimeoutSec           = 60
    }
}

function Invoke-DNSTest {
    param([hashtable]$Context)
    $cfg     = Get-DefaultConfig
    $sources = $Context.Config.windows_dns_sources
    $results = [System.Collections.Generic.List[psobject]]::new()

    foreach ($src in $sources) {
        foreach ($zone in $src.zones) {

            # ── AXFR: Full zone transfer ──────────────────────────
            Write-Host "  [MOD-10] AXFR $zone from $($src.name) ($($src.ip))"
            $sw       = [System.Diagnostics.Stopwatch]::StartNew()
            $axfrOut  = & dig "@$($src.ip)" $zone AXFR +time=$($cfg.TimeoutSec) 2>&1
            $sw.Stop()

            $axfrLines   = $axfrOut | Where-Object { $_ -notmatch '^;' -and $_.Trim() }
            $recordCount = $axfrLines.Count
            $axfrOk      = $axfrOut -match 'Transfer complete' -or $recordCount -gt 2
            $axfrRefused = $axfrOut -match 'REFUSED|XFR size'

            $results.Add([pscustomobject]@{
                Source       = $src.name
                Zone         = $zone
                TransferType = 'AXFR'
                DurationMs   = [math]::Round($sw.Elapsed.TotalMilliseconds, 2)
                RecordCount  = $recordCount
                Refused      = if ($axfrRefused) { $true } else { $false }
                Status       = if ($axfrOk)      { 'PASS' }
                              elseif ($axfrRefused) { 'FAIL' }
                              else                   { 'WARN' }
                Detail       = if ($axfrOk) { "$recordCount records transferred" } else { "Transfer incomplete or REFUSED" }
            })

            # ── SOA serial check — compare primary to expected ────
            $soaOut   = & dig "@$($src.ip)" $zone SOA +short 2>&1
            $serial   = if ($soaOut -match '\s(\d{10})\s') { [long]$Matches[1] } else { 0 }
            $results.Add([pscustomobject]@{
                Source       = $src.name
                Zone         = $zone
                TransferType = 'SOA-Serial'
                Serial       = $serial
                Status       = if ($serial -gt 0) { 'PASS' } else { 'FAIL' }
                Detail       = "SOA serial: $serial"
            })

            # ── IXFR: Incremental transfer (requires known serial) ─
            if ($cfg.TransferTypes -contains 'IXFR' -and $serial -gt 0) {
                $prevSerial = $serial - 1    # ask for changes since serial-1
                Write-Host "  [MOD-10] IXFR $zone serial=$prevSerial → $serial"
                $ixfrOut  = & dig "@$($src.ip)" $zone IXFR=$prevSerial +time=$($cfg.TimeoutSec) 2>&1
                $ixfrOk   = $ixfrOut -match 'SOA'
                $results.Add([pscustomobject]@{
                    Source=$src.name; Zone=$zone; TransferType='IXFR'
                    Status = if ($ixfrOk) { 'PASS' } else { 'WARN' }
                    Detail = if ($ixfrOk) { "Incremental transfer responded with SOA" } else { "IXFR fell back to AXFR or failed" }
                })
            }

            # ── ACL blocking test — unauthorized source should get REFUSED ─
            if ($cfg.TestAclBlocking) {
                # Use a source-routed probe via nsupdate trick or flag for manual verification
                $results.Add([pscustomobject]@{
                    Source       = $src.name
                    Zone         = $zone
                    TransferType = 'ACL-Block'
                    Status       = 'MANUAL'
                    Detail       = "Verify AXFR REFUSED from non-ACL IP via: dig @$($src.ip) $zone AXFR +tcp"
                })
            }
        }
    }

    # ── Split-DNS consistency: internal vs external answers differ ──
    $splitZones = $sources | Where-Object { $_.role -match 'internal|external' } |
        Select-Object -ExpandProperty zones -Unique | Group-Object |
        Where-Object { $_.Count -gt 1 }    # zones present on both int + ext

    foreach ($zg in $splitZones) {
        $zName   = $zg.Name
        $intSrc  = $sources | Where-Object { $_.role -match 'internal' -and $_.zones -contains $zName } | Select-Object -First 1

    }

    $logApi       = $feeds.sinkhole_log_api

                          elseif ($r.Status -eq 'NXDOMAIN')           { 'NXDOMAIN'    }

                Test       = 'Bypass-Isolation'

                -ContentType 'application/json' -ErrorAction SilentlyContinue

# Helper: writes an nsupdate batch file and executes it, returning stdout+stderr
function Invoke-NSUpdate {
    param(
        [string] $Server,
        [string] $KeyName,
        [string] $KeyAlgorithm,
        [string] $KeySecret,
        [string] $Zone,
        [string[]]$UpdateLines,     # e.g. 'update add foo.zone. 60 A 1.2.3.4'
        [switch] $NoTSIG
    )
    $batchFile = [System.IO.Path]::GetTempFileName()
    $content   = [System.Collections.Generic.List[string]]::new()
    $content.Add("server $Server")
    if (-not $NoTSIG) {
        $content.Add("key $KeyAlgorithm:$KeyName $KeySecret")
    }
    $content.Add("zone $Zone")
    $UpdateLines | ForEach-Object { $content.Add($_) }
    $content.Add("send")
    $content | Set-Content $batchFile

    $sw  = [System.Diagnostics.Stopwatch]::StartNew()
    $out = & nsupdate $batchFile 2>&1
    $sw.Stop()
    Remove-Item $batchFile -Force -ErrorAction SilentlyContinue

    [pscustomobject]@{
        Output     = $out
        LatencyMs  = [math]::Round($sw.Elapsed.TotalMilliseconds, 2)
        Success    = -not ($out -match 'REFUSED|NOTAUTH|SERVFAIL|timed out|bad')
        Refused    = $out -match 'REFUSED|NOTAUTH'
    }
}

# Resolve full CNAME chain using dig +trace-style multi-query
function Resolve-CNAMEChain($Server, $Name, $ECSSubnet, $MaxDepth) {
    $chain  = [System.Collections.Generic.List[string]]::new()
    $current = $Name
    $finalIP = $null

    for ($i=0; $i-lt$MaxDepth; $i++) {
        $digArgs  = @("@$Server", $current, 'A', '+time=3', '+tries=1')
        if ($ECSSubnet) { $digArgs += "+subnet=$ECSSubnet" }
        $out = & dig @digArgs 2>&1

        # Parse ANSWER section
        $cnameMatch = $out | Where-Object { $_ -match '\s+CNAME\s+' } | Select-Object -First 1
        $aMatch     = $out | Where-Object { $_ -match '\s+A\s+\d'    } | Select-Object -First 1

        if ($aMatch -match '\s+A\s+([\d.]+)') {
            $chain.Add($current)
            $finalIP = $Matches[1]
            break
        } elseif ($cnameMatch -match '\s+CNAME\s+(\S+)') {
            $chain.Add($current)
            $current = $Matches[1].TrimEnd('.')
        } else {
            break
        }
    }

    [pscustomobject]@{ Chain=$chain; FinalIP=$finalIP; Depth=$chain.Count }
}

# Match an IP to a cloud provider based on configured prefixes
function Get-CloudProvider([string]$IP, $CloudPrefixes) {
    foreach ($csp in $CloudPrefixes.PSObject.Properties) {
        foreach ($prefix in $csp.Value) {
            try {
                $net    = [System.Net.IPNetwork]::Parse($prefix)
                $ipAddr = [System.Net.IPAddress]::Parse($IP)
                if ($net.Contains($ipAddr)) { return $csp.Name }
            } catch {}
        }
    }
    return 'unknown'
}

HTML Report Engine — MOD-09
Reads raw JSON output from each module's run directory and generates a dark-theme HTML dashboard with module summary, per-target latency table, and new v2.0 security/zone panels.
The report engine auto-detects which modules ran from files present in RunDir/raw/*.json. Adding new modules requires no changes to the report engine — all modules surface their Summary.Status and Summary.Message automatically.
DNSTest-Report.psm1 — MOD-09 (v2.0)

function Get-ModuleInfo {
    [pscustomobject]@{ Id='MOD-09'; Name='HTML Report Engine'; Version='2.0'
        Tags=@('report','output','html') }
}
function Get-DefaultConfig { @{ OutputFile='index.html' } }
function Invoke-DNSTest {
    param([hashtable]$Context)
    Invoke-DNSTestReport -RunDir $Context.RunDir
    return [pscustomobject]@{
        ModuleId='MOD-09'
        Summary=[pscustomobject]@{ Status='PASS'; TestCount=1; Message='HTML report generated' }

function Get-StatusBadge($s) {
    switch($s) {
        'PASS'   { "<span style='color:#22c55e;font-weight:700'>● PASS</span>"   }
        'WARN'   { "<span style='color:#f59e0b;font-weight:700'>◐ WARN</span>"   }
        'FAIL'   { "<span style='color:#ef4444;font-weight:700'>✕ FAIL</span>"   }
        'MANUAL' { "<span style='color:#89b4d1;font-weight:700'>◻ MANUAL</span>" }
        default  { "<span style='color:#5a6e87'>$s</span>"                         }
    }
}

function Invoke-DNSTestReport([string]$RunDir) {

Runner — Invoke-DNSTestSuite.ps1
Entry point. Accepts scenario name or explicit module list. Supports dry-run and no-report modes.
Invoke-DNSTestSuite.ps1 
# ============================================================
# Invoke-DNSTestSuite.ps1 — DNSTestToolkit v2.0 Entry Point
# ============================================================
param(
    [string]    $Scenario  = 'smoke',
    [string[]]  $Modules   = @('*'),
    [switch]    $NoReport,
    [switch]    $DryRun,
    [switch]    $ListScenarios
)

. (Join-Path $PSScriptRoot 'DNSTestEngine.ps1')

if ($ListScenarios) {
    $sc = Get-EngineConfig 'scenarios'
    $sc.scenarios.PSObject.Properties | ForEach-Object {
        Write-Host "  $($_.Name.PadRight(20)) ~$($_.Value.estimated_minutes)min  $($_.Value.description)"
    }
    return
}

if ($Modules -eq @('*') -and $Scenario) {
    $sc       = Get-EngineConfig 'scenarios'
    $scDef    = $sc.scenarios.$Scenario
    if ($scDef) {
        $Modules = $scDef.modules
        Write-Host "[SUITE] Scenario: $Scenario | $($scDef.description)" -ForegroundColor Cyan
        Write-Host "[SUITE] Estimated runtime: ~$($scDef.estimated_minutes) minutes"  -ForegroundColor Cyan
        if ($scDef.notes) { Write-Host "[SUITE] NOTE: $($scDef.notes)" -ForegroundColor Yellow }
    } else {
        Write-Warning "Unknown scenario '$Scenario'. Run with -ListScenarios to see options."
        return
    }
}

Invoke-TestEngine -Modules $Modules -Scenario $Scenario `
    -NoReport:$NoReport -DryRun:$DryRun
